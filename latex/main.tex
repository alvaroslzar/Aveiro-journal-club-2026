\section{Motivation}
\begin{frame}{Motivation}
    The McVittie solution\footcite{McVittie:1933zz}
    \begin{align}
    ds^2 = -\left( \frac{1-\frac{M}{2a(t)R}}{1+\frac{M}{2a(t)R}} \right)^2 dt^2 + \left( 1+\frac{M}{2a(t)R} \right)^4 a^2(t) \left\{ dR^2 +R^2\,d\Omega^2 \right\}
    \end{align}
    describes a massive object embedded in a FLRW Universe.

    \begin{itemize}
    \item <2-> Recently, it has been shown\footcite{Modesto:2025cre} that there is a \textbf{naked curvature singularity at the horizon}.

    \item<3-> Is this exlusive of the McVittie solution?

    \item<4-> Can this be avoided? If so, how?
    \end{itemize}
\end{frame}


\section{Singularity problems}
\begin{frame}{Singularity problems}
    Indeed, this is a \textbf{rather universal phenomenon}, that is, the horizon embedding in a dynamical background becomes a naked singularity for a large class of models.

    We consider the following form of the metric a the \textbf{dynamical} and \textbf{spherically symmetric spacetime}
    \begin{align}
    ds^2 = - \e^{2\nu (t,r)}\, dt^2 + \e^{2\lambda (t,r)}\, dr^2 + \e^{2\xi (t,r)}\,d\Omega^2 , 
    \end{align}

    \begin{itemize}
        \item<2-> At $\e^{2\nu (t,r)}=0$, we have the horizon(s).
        \item<2-> There, \textbf{curvatures are singular in general}.
    \end{itemize}
\end{frame}


\begin{frame}
    The Ricci scalar:
    \only<1>{
        \begin{align*}
        R= 2 \e^{-2 \nu} \left\{\ddot\lambda + \left( \dot\lambda - \dot\nu \right) \dot\lambda \right\} + 4 \e^{-2\nu} \left\{ \ddot\xi + \left(\dot\xi - \dot\nu \right) \dot \xi \right\} + ...
        \end{align*}
    }
    \onslide<2->{
        \begin{align*} % Same but with underbraces
        R= 2 \e^{-2 \nu} 
        \underbrace{ \left\{\ddot\lambda + \left( \dot\lambda - \dot\nu \right) \dot\lambda \right\}}_{\text{vanish at }\e^{2\nu}=0} + 4 \e^{-2\nu} \underbrace{\left\{ \ddot\xi + \left(\dot\xi - \dot\nu \right) \dot \xi \right\}}_{\text{vanish at }\e^{2\nu}=0} + ...
        \end{align*}
    }
    \onslide<3->{
        We further require that:
        \begin{itemize}
            \item $\e^{2\lambda+2\nu}$ is neither singular nor vanishes at the horizon.

            \item $\e^\xi$ is neither singular nor vanishes at the horizon.
        \end{itemize}
    }
\end{frame}


\begin{frame}
    \onslide<1->{
        Then, the metric functions can be calculated as
        \begin{align}
        \e^{\lambda (t,r)} =&\, \int^t dt_1 \e^{\nu\left(t_1, r\right)} \int^{t_1} dt_2 \Lambda \left(t_2, r\right)\, \\
        \label{r4}
        \e^{\xi (t,r)} =&\, \int^t dt_1 \e^{\nu\left(t_1, r\right)} \int^{t_1} dt_2 \Xi \left(t_2, r\right) \e^{\nu\left(t_2, r\right)} \,.
        \end{align}
    }
    \onslide<2->{
        Note that, for $\e^{\nu}=\e^{\nu (r)}$, the areal radius reads
        \begin{align}
        \label{xi1}
        \e^{\xi\left(t,r\right)} = \Xi_1 (r) + \e^{2\nu(r)} \Xi_2(r,t)\, , \quad \Xi_2(r,t) \equiv \int^t dt_1 \int^{t_1} dt_2 \Xi \left(t_2, r\right) \, .
        \end{align}
    }
\end{frame}


\subsection{Examples}
\begin{frame}{Examples: Simpson-Visser black bounce}
    We consider \textbf{exotic compact objects}, such as traversable wormholes and black holes with a throat.
    \onslide<2->{
        \begin{itemize}
            \item First, the time-dependent extension of the Simpson-Visser black bounce\footcite{Simpson:2018tsi}, whose metric is given by
            \begin{align}
            \e^{2\nu}=\e^{-2\lambda}= 1 - \frac{2M}{\sqrt{r^2 + {r_\mathrm{throat}}^2}}\, , \quad 
            r\in (-\infty, \infty) .
            \end{align}
            Choosing the functions in \eqref{xi1} as $\Xi_1(r)=\sqrt{r^2+{\rt}^2}$ and $\Xi_2(t,r)=e^{-2\nu(r)}\Xi_{02}(t)$, we get the areal radius function
            \begin{align}
            \e^\xi = \sqrt{r^2+{\rt}^2}+\Xi_{02}(t)\,.
            \end{align}
            For $r>0$, this geometry presents one event horizon at $r=\sqrt{4-{\rt}^2}$ when $\rt<2M$.
        \end{itemize}
    }
\end{frame}


\begin{frame}{Examples: Hayward-like wormhole}
    \begin{itemize}
        \item Next, we consider the time-dependent extension of the Hayward geometry\footcite{Hayward:2005gi}:
        \begin{align}
        \label{Hayward}
        \e^{2\nu}=\e^{-2\lambda}=1 - \frac{2Mr^2}{\left|r\right|^3 + 2M\gamma^2} \, , \quad 
        r\in (-\infty, \infty) .
        \end{align}
        Similarly, choosing the functions $\Xi_1(r)=\sqrt{r^2+{\rt}^2}$ and $\Xi_2(t,r)=e^{-2\nu(r)}\Xi_{02}(t)$  in \eqref{xi1}, the areal radius function reads
        \begin{align}
        \e^\xi = \sqrt{r^2+{\rt}^2}+\Xi_{02}(t)\,.
        \end{align}
        For $r>0$, this geometry presents two event horizons when $\gamma<4\sqrt{3}M/9\approx 0.770M$.
    \end{itemize}
\end{frame}


\begin{frame}
    Bottom line:
    \begin{itemize}
        \item<1-> When we consider a black hole in a dynamical background, the horizon becomes a naked singularity for a large class of models.

        \item<2-> We give general prescriptions on how this can be avoided.

        \item<3-> As examples, we construct the dynamical extensions of the Simpson-Visser black bounce and a Hayward-like wormhole.
        In these particular cases, \textbf{only $\e^\xi$ depends on $t$}.
    \end{itemize}
\end{frame}


\section{Shadows of non-stationary wormholes}
\begin{frame}{Shadows of non-stationary wormholes}
    \onslide<1->{
        In order to simulate their optical appearances, we need to solve the \textbf{null geodesic equations}.
        In the plane $\vartheta=\pi/2$ :
        \begin{align}
        \frac{d}{d\sigma}\left(\e^{2\nu} k^t \right) + \frac{(k^\varphi)^2}{2}\frac{\partial}{\partial t}\left( \e^{2\xi} \right) = 0\, ,
        \quad
        \frac{d}{d\sigma}\left( \e^{2\xi}k^\varphi \right) = 0\, ,
        \end{align}
    }
    \onslide<2->{
        Using the \textbf{adiabatic approximation} $\implies \partial_t \e^\xi \approx 0$.
        We shall treat $\Xi_{02}(t)\equiv\Xi$ as an additional parameter, so we get two conserved quantities:
        \begin{align}
        \e^{2\nu} k^t = E\, ,
        \quad
        \e^{2\xi} k^\varphi = L\, .
        \end{align}
    }
\end{frame}


\begin{frame}
    \onslide<1->{
        Null geodesics only depend on their \textbf{impact parameter} $b=L/E$ and the \textbf{effective potential} $V(r)=e^{2\nu-2\xi}$
        \begin{align}
        \left(\frac{d r}{d\sigma}\right)^2 = \frac{1}{b^2} - V(r)\,.
        \end{align}
    }
    \onslide<2->{
        The (possibly several) maxima of the effective potential $V(r)$ are called \textbf{photon spheres} $\rp$, which define the so-called \textbf{critical impact parameters}
        \begin{align}
            b_c = \frac{1}{\sqrt{V(\rp)}}\,.
        \end{align}
        Geodesics approaching one of those $b_c$ will turn an arbitrarily large number of times around the compact object in unstable circular orbits.
    }
\end{frame}


\subsection{Ray-tracing method}
\begin{frame}{Ray-tracing method}
    We know simulate the shadow of the previous examples using the so-called \textbf{ray-tracing method}\footcite{Gralla:2019xty,Guerrero:2021ues}.

    Assumptions and simplifications:
    \begin{itemize}
        \item<2-> The emission is only due to the accretion disk outside of the throat.

        \item<3-> The accretion disk is both geometrically and optically thin.
        
        \item<4-> The disk is lying on the equatorial plane and it is observed from a \textbf{face on} view.
        
        \item<5-> The disk is \textbf{static}, that is, the redshift is only due to gravitational redshift.
        
        \item<6-> The emission is isotropic in the rest frame of the disk.
    \end{itemize}
\end{frame}


\begin{frame}
    \begin{enumerate}
        \item[1.] Compute the trajectory of the geodesic
    \end{enumerate}
    \begin{equation}
    \frac{d\varphi}{dr} = \pm \frac{b\ \e^{-\xi}}{\sqrt{\e^{2\xi}-b^2 \e^{2\nu}}} \,.
    \label{eq:trajectory}
    \end{equation}

\begin{frame}
    \begin{enumerate}
        \item[2.] A geodesic with impact parameter $b$, has its $m$-th intersection with the disk at some radial coordinate $r$ .
        This information is stored in the \textbf{transfer functions} $r_m(b)$ .
    \end{enumerate}

    \begin{figure}
    \includegraphics{figures/method/transfer_function.pdf}
    \end{figure}
\end{frame}


\begin{frame}
    \begin{enumerate}
        \item[3.]<1-> The disk emits with \textbf{specific intensity} $I^\mathrm{em}_\omega$ in its rest frame.
        Since $I^\mathrm{em}_\omega/\omega^3$ is conserved\footnote{This may be derived from Liouville's theorem.} along any given geodesic
        \begin{align}
        I^\mathrm{ob}_{\omega'} = \left(\frac{\omega'}{\omega}\right)^3 I^\mathrm{em}_\omega = \e^{3\nu}I^\mathrm{em}_\omega \implies I^\mathrm{ob} = \int I^\mathrm{ob}_{\omega'} d\omega' = \e^{4\nu}I^\mathrm{em}\, .
        \end{align}

        \item[4.]<2-> The \textbf{observed intensity} profile is then the sum of all intersections with the disk
        \begin{align}
        I^\mathrm{ob}(b) = \sum_{m=1}^\infty \left. \e^{4\nu(r)}I^\mathrm{em}(r)\right|_{r=r_m(b)}\, .
        \label{eq:observed-intensity}
        \end{align}
        We truncate the sum at $m=3$, since higher-order intersections contribute negligibly to the total intensity\footcite{Gralla:2019xty, Bisnovatyi-Kogan:2022ujt}.
    \end{enumerate}
\end{frame}


\begin{frame}
    \begin{itemize}
        \item[5.] To model the emission, we use the parametrised family of toy-models\footcite{Gralla:2020srx, Paugnat:2022qzy, Cardenas-Avendano:2022csp}
        \begin{align}
        I^\mathrm{em}(r;\mu,\sigma,\beta) = \frac{\exp \left( -\frac{1}{2}\left(\beta+\arcsinh\left(\frac{r-\mu}{\sigma}\right)\right)^2 \right)}{ \sqrt{(r-\mu)^2+\sigma^2} }\, .
        \label{eq:su-model}
        \end{align}
    \end{itemize}

\end{frame}

\begin{frame}{Singularity problems}
    The Ricci scalar:
    \begin{align*}
    R= 2 \e^{-2 \nu} \left\{\ddot\lambda + \left( \dot\lambda - \dot\nu \right) \dot\lambda \right\} + 4 \e^{-2\nu} \left\{ \ddot\xi + \left(\dot\xi - \dot\nu \right) \dot \xi \right\} + ...
    \end{align*}

    For the Ricci scalar to be regular, we require
    \begin{align}
    \ddot\lambda + \left( \dot\lambda - \dot\nu \right) \dot\lambda =&\, \Lambda_0 \left(t, r\right) \e^{2\nu}\, , \\
    \ddot\xi + \left(\dot\xi - \dot\nu \right) \dot \xi =&\, \Xi_0 \left(t, r\right) \e^{2\nu}\,, 
    \end{align}
    where $\Lambda_0\left(t, r\right)$ and $\Xi_0\left(t, r\right)$ are smooth.
\end{frame}


\begin{frame}{Time-independent horizon}
    \begin{itemize}
        \item The metric function $\e^\lambda$ can be calculated as
        \begin{align}
        \e^{\lambda (t,r)} =&\, \int^t dt_1 \e^{\nu\left(t_1, r\right)} \int^{t_1} dt_2 \Lambda \left(t_2, r\right)\,.
        \end{align}

        \item $\e^\lambda$ is singular at $\e^\nu=0$, which is due to the integration ``constants" w.r.t. $t_1$ and $t_2$.
        
        \item Since those integration ``constants" are only functions of $r$, the horizon is time-independent.
        \end{itemize}
\end{frame}
